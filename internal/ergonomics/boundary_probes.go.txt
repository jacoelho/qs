// Executable architecture probes against the frozen pre-change library.
// These adapters test caller boundaries, not a proposed production implementation.
package main

import (
	"errors"
	"fmt"
	"reflect"
	"slices"

	"github.com/jacoelho/qs"
)

type renderable interface{ ToSQL() (string, []any, error) }
type input struct {
	columns []string
	row     []qs.WriteValue
	keys    []string
	updates []qs.Assignment
}

var errEmpty = errors.New("explicit input collection is empty")

// BEGIN callerA
func callerA(in input) (renderable, error) {
	if len(in.keys) == 0 {
		return nil, errEmpty
	}
	return qs.InsertInto("documents").ColumnsSlice(in.columns).ValuesSlice(in.row).
		OnConflict(qs.ConflictColumns(in.keys[0], in.keys[1:]...).DoUpdateSlice(in.updates)), nil
}

// END callerA

// BEGIN callerB
func callerB(in input) (renderable, error) {
	return bInsert("documents").Columns(in.columns...).Values(in.row...).
		OnConflict(bConflict(in.keys...).DoUpdate(in.updates...)), nil
}

// END callerB

// BEGIN adapterB
type bTarget struct{ target qs.InsertTarget }
type bColumns struct{ target qs.InsertColumnsTarget }
type bRows struct {
	rows *qs.InsertRows
	err  error
}
type bConflictTarget struct{ keys []string }
type bAction struct {
	action qs.ConflictUpdate
	err    error
}

func bInsert(table string) bTarget { return bTarget{qs.InsertInto(table)} }
func (b bTarget) Columns(columns ...string) bColumns {
	return bColumns{b.target.ColumnsSlice(columns)}
}
func (b bColumns) Values(row ...qs.WriteValue) bRows {
	return bRows{rows: b.target.ValuesSlice(row)}
}
func bConflict(keys ...string) bConflictTarget { return bConflictTarget{slices.Clone(keys)} }
func (b bConflictTarget) DoUpdate(updates ...qs.Assignment) bAction {
	if len(b.keys) == 0 {
		return bAction{err: errEmpty}
	}
	return bAction{action: qs.ConflictColumns(b.keys[0], b.keys[1:]...).DoUpdateSlice(updates)}
}
func (b bRows) OnConflict(action bAction) bRows {
	if action.err != nil {
		b.err = action.err
		return b
	}
	b.rows.OnConflict(action.action)
	return b
}
func (b bRows) ToSQL() (string, []any, error) {
	if b.err != nil {
		return "", nil, b.err
	}
	return b.rows.ToSQL()
}

// END adapterB

// BEGIN callerC
func callerC(in input) (renderable, error) {
	columns, err := nonEmpty(in.columns)
	if err != nil {
		return nil, err
	}
	row, err := nonEmpty(in.row)
	if err != nil {
		return nil, err
	}
	keys, err := nonEmpty(in.keys)
	if err != nil {
		return nil, err
	}
	updates, err := nonEmpty(in.updates)
	if err != nil {
		return nil, err
	}
	return cUpsert(columns, row, keys, updates)
}

// END callerC

// BEGIN adapterC
// The zero value is invalid; construction validates only cardinality.
type nonEmptyList[T any] struct{ values []T }

func nonEmpty[T any](values []T) (nonEmptyList[T], error) {
	if len(values) == 0 {
		return nonEmptyList[T]{}, errEmpty
	}
	return nonEmptyList[T]{slices.Clone(values)}, nil
}
func cUpsert(columns nonEmptyList[string], row nonEmptyList[qs.WriteValue], keys nonEmptyList[string], updates nonEmptyList[qs.Assignment]) (renderable, error) {
	if len(columns.values) == 0 || len(row.values) == 0 || len(keys.values) == 0 || len(updates.values) == 0 {
		return nil, errEmpty
	}
	return qs.InsertInto("documents").ColumnsSlice(columns.values).ValuesSlice(row.values).
		OnConflict(qs.ConflictColumns(keys.values[0], keys.values[1:]...).DoUpdateSlice(updates.values)), nil
}

// END adapterC

// BEGIN callerD
func callerD(in input) (renderable, error) {
	return (upsertSpec{Table: "documents", Columns: in.columns, Row: in.row,
		ConflictKeys: in.keys, Updates: in.updates}).Build()
}

// END callerD

// BEGIN adapterD
// This is a mutable input spec, not an immutable command. Build captures slices.
// Columns are mandatory for this narrow operation, so omitted means invalid.
type upsertSpec struct {
	Table        string
	Columns      []string
	Row          []qs.WriteValue
	ConflictKeys []string
	Updates      []qs.Assignment
}

func (s upsertSpec) Build() (renderable, error) {
	if len(s.Columns) == 0 || len(s.Row) == 0 || len(s.ConflictKeys) == 0 || len(s.Updates) == 0 {
		return nil, errEmpty
	}
	return qs.InsertInto(s.Table).ColumnsSlice(s.Columns).ValuesSlice(s.Row).
		OnConflict(qs.ConflictColumns(s.ConflictKeys[0], s.ConflictKeys[1:]...).DoUpdateSlice(s.Updates)), nil
}

// END adapterD

func fixture() input {
	return input{columns: []string{"id", "title"}, row: []qs.WriteValue{qs.Value(7), qs.Value("draft")},
		keys: []string{"id"}, updates: []qs.Assignment{qs.Set("title", qs.Value("published"))}}
}

const expectedSQL = `INSERT INTO "documents" ("id", "title") VALUES ($1, $2) ON CONFLICT ("id") DO UPDATE SET "title" = $3`

var expectedArgs = []any{7, "draft", "published"}

func render(q renderable, err error) (string, []any, error) {
	if err != nil {
		return "", nil, err
	}
	return q.ToSQL()
}
func assertValid(name string, q renderable, err error) {
	sql, args, err := render(q, err)
	if err != nil || sql != expectedSQL || !reflect.DeepEqual(args, expectedArgs) {
		panic(fmt.Sprintf("%s: sql=%q args=%#v err=%v", name, sql, args, err))
	}
}
func assertInvalid(name string, q renderable, err error) {
	sql, args, err := render(q, err)
	if err == nil || sql != "" || len(args) != 0 {
		panic(fmt.Sprintf("%s: invalid input escaped: %q %#v %v", name, sql, args, err))
	}
}

func main() {
	callers := []struct {
		name  string
		build func(input) (renderable, error)
	}{
		{"A current staged", callerA}, {"B uniform variadic adapter", callerB},
		{"C validated nonempty", callerC}, {"D operation spec", callerD},
	}
	for _, c := range callers {
		q, err := c.build(fixture())
		assertValid(c.name+" valid", q, err)
		in := fixture()
		q, err = c.build(in)
		in.columns[0], in.row[0], in.keys[0], in.updates[0] = "mutated", qs.Value(99), "mutated", qs.Set("mutated", qs.Value(99))
		assertValid(c.name+" captured input positions", q, err)
		for _, mode := range []string{"nil columns", "empty columns", "nil row", "empty row", "nil keys", "empty keys", "nil updates", "empty updates", "invalid element", "mismatched row width"} {
			in = fixture()
			switch mode {
			case "nil columns":
				in.columns = nil
			case "empty columns":
				in.columns = []string{}
			case "nil row":
				in.row = nil
			case "empty row":
				in.row = []qs.WriteValue{}
			case "nil keys":
				in.keys = nil
			case "empty keys":
				in.keys = []string{}
			case "nil updates":
				in.updates = nil
			case "empty updates":
				in.updates = []qs.Assignment{}
			case "invalid element":
				in.updates[0] = qs.Assignment{}
			case "mismatched row width":
				in.row = in.row[:1]
			}
			q, err = c.build(in)
			assertInvalid(c.name+" "+mode, q, err)
		}
		fmt.Printf("PASS %s: exact SQL/args; captured positions; 8 empty collections; invalid element; row width (12 probes)\n", c.name)
	}
	q, err := cUpsert(nonEmptyList[string]{}, nonEmptyList[qs.WriteValue]{}, nonEmptyList[string]{}, nonEmptyList[qs.Assignment]{})
	assertInvalid("C zero lists", q, err)
	fmt.Println("PASS C zero-value nonempty lists rejected (1 probe)")
	fmt.Println("49 boundary probes passed; no timing or allocation claim")
}
