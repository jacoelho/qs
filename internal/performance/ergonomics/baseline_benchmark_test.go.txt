package qs_test

import (
	"reflect"
	"testing"

	"github.com/jacoelho/qs"
	"github.com/jacoelho/qs/projection"
)

var (
	architectureStatement  qs.Statement
	architectureSQL        string
	architectureBytes      []byte
	architectureArgs       []any
	architectureProjection projection.Projection[string]
	architectureMetadata   []string
)

func architectureDynamicUpsert(columns, keys []string, values []qs.WriteValue, updates []qs.Assignment) qs.Statement {
	return qs.InsertInto("users").ColumnsSlice(columns).ValuesSlice(values).
		OnConflict(qs.ConflictColumns(keys[0], keys[1:]...).DoUpdateSlice(updates)).Returning(qs.Col("id"))
}

func architectureLiteralUpsert() qs.Statement {
	return qs.InsertInto("users").Columns("tenant_id", "id", "email", "payload").
		Values(qs.Value(7), qs.Value(42), qs.Value("ada@example.test"), qs.Write(qs.Param(`{"active":true}`).Cast(qs.TypeJSONB))).
		OnConflict(qs.ConflictColumns("tenant_id", "id").DoUpdate(
			qs.Set("email", qs.Write(qs.Excluded("email"))), qs.Set("payload", qs.Write(qs.Excluded("payload"))),
		)).Returning(qs.Col("id"))
}

func architectureExpressionTargets(targets []qs.Expr, values []qs.WriteValue) qs.Statement {
	return qs.InsertInto("settings").Targets(targets[0], targets[1:]...).ValuesSlice(values).Returning(qs.Star())
}

func architectureSparsePatch(assignments []qs.Assignment) qs.Statement {
	if len(assignments) == 0 {
		return nil
	}
	return qs.Update("users").Set(assignments...).Where(qs.Eq("id", 42))
}

func architectureConcat(left, right projection.Projection[string]) projection.Projection[string] {
	expressions, metadata := right.Expressions(), right.Metadata()
	columns := make([]projection.Column[string], len(expressions))
	for i := range expressions {
		columns[i] = projection.Output(expressions[i], metadata[i])
	}
	return left.With(columns...)
}

func architectureCheck(b *testing.B, q qs.Statement, wantSQL string, wantArgs []any) {
	b.Helper()
	sql, args, err := q.ToSQL()
	if err != nil || sql != wantSQL || !reflect.DeepEqual(args, wantArgs) {
		b.Fatalf("fixture: SQL=%q args=%#v error=%v; want %q %#v", sql, args, err, wantSQL, wantArgs)
	}
}

func BenchmarkArchitectureQueries(b *testing.B) {
	columns := []string{"tenant_id", "id", "email", "payload"}
	keys := []string{"tenant_id", "id"}
	values := []qs.WriteValue{qs.Value(7), qs.Value(42), qs.Value("ada@example.test"), qs.Write(qs.Param(`{"active":true}`).Cast(qs.TypeJSONB))}
	updates := []qs.Assignment{qs.Set("email", qs.Write(qs.Excluded("email"))), qs.Set("payload", qs.Write(qs.Excluded("payload")))}
	targets := []qs.Expr{qs.Col("id"), qs.Col("payload").Field("active")}
	targetValues := []qs.WriteValue{qs.Value(42), qs.Value(true)}
	const upsertSQL = `INSERT INTO "users" ("tenant_id", "id", "email", "payload") VALUES ($1, $2, $3, ($4)::jsonb) ON CONFLICT ("tenant_id", "id") DO UPDATE SET "email" = "excluded"."email", "payload" = "excluded"."payload" RETURNING "id"`
	upsertArgs := []any{7, 42, "ada@example.test", `{"active":true}`}
	for _, fixture := range []struct {
		name  string
		build func() qs.Statement
		sql   string
		args  []any
	}{
		{"DynamicUpsert", func() qs.Statement { return architectureDynamicUpsert(columns, keys, values, updates) }, upsertSQL, upsertArgs},
		{"LiteralUpsert", architectureLiteralUpsert, upsertSQL, upsertArgs},
		{"ExpressionTargets", func() qs.Statement { return architectureExpressionTargets(targets, targetValues) }, `INSERT INTO "settings" ("id", "payload"."active") VALUES ($1, $2) RETURNING *`, []any{42, true}},
	} {
		b.Run(fixture.name, func(b *testing.B) {
			q := fixture.build()
			architectureCheck(b, q, fixture.sql, fixture.args)
			b.Run("Construct", func(b *testing.B) {
				b.ReportAllocs()
				for b.Loop() {
					architectureStatement = fixture.build()
				}
			})
			b.Run("ConstructAndToSQL", func(b *testing.B) {
				b.ReportAllocs()
				for b.Loop() {
					var err error
					architectureSQL, architectureArgs, err = fixture.build().ToSQL()
					if err != nil {
						b.Fatal(err)
					}
				}
			})
			b.Run("AppendWarm", func(b *testing.B) {
				sql := make([]byte, 0, len(fixture.sql))
				args := make([]any, 0, len(fixture.args))
				b.ReportAllocs()
				for b.Loop() {
					var err error
					sql, args, err = q.AppendSQL(sql[:0], args[:0])
					if err != nil {
						b.Fatal(err)
					}
				}
				architectureBytes, architectureArgs = sql, args
			})
		})
	}
}

func BenchmarkArchitectureProjection(b *testing.B) {
	left := projection.New(
		projection.Named(qs.Col("id"), "id"), projection.Named(qs.Col("tenant_id"), "tenant_id"),
		projection.Named(qs.Col("kind"), "kind"), projection.Named(qs.Col("state"), "state"),
		projection.Named(qs.Col("attempt"), "attempt"), projection.Named(qs.Col("created_at"), "created_at"),
		projection.Named(qs.Col("scheduled_at"), "scheduled_at"), projection.Named(qs.Col("priority"), "priority"),
	)
	right := projection.New(
		projection.Named(qs.Col("name"), "name"), projection.Named(qs.Col("email"), "email"),
		projection.Named(qs.Col("active"), "active"), projection.Named(qs.Col("payload"), "payload"),
		projection.Named(qs.Col("queue"), "queue"), projection.Named(qs.Col("locked_at"), "locked_at"),
		projection.Named(qs.Col("updated_at"), "updated_at"), projection.Named(qs.Col("version"), "version"),
	)
	const wantSQL = `SELECT "id" AS "id", "tenant_id" AS "tenant_id", "kind" AS "kind", "state" AS "state", "attempt" AS "attempt", "created_at" AS "created_at", "scheduled_at" AS "scheduled_at", "priority" AS "priority", "name" AS "name", "email" AS "email", "active" AS "active", "payload" AS "payload", "queue" AS "queue", "locked_at" AS "locked_at", "updated_at" AS "updated_at", "version" AS "version"`
	wantMetadata := []string{"id", "tenant_id", "kind", "state", "attempt", "created_at", "scheduled_at", "priority", "name", "email", "active", "payload", "queue", "locked_at", "updated_at", "version"}
	p := architectureConcat(left, right)
	architectureCheck(b, qs.Select(p.Expressions()...), wantSQL, nil)
	if !reflect.DeepEqual(p.Metadata(), wantMetadata) {
		b.Fatal("composed projection metadata changed")
	}
	b.Run("Concat", func(b *testing.B) {
		b.ReportAllocs()
		for b.Loop() {
			architectureProjection = architectureConcat(left, right)
		}
	})
	b.Run("ConcatAndSelect", func(b *testing.B) {
		b.ReportAllocs()
		for b.Loop() {
			p := architectureConcat(left, right)
			architectureStatement = qs.Select(p.Expressions()...)
			architectureMetadata = p.Metadata()
		}
	})
}

func BenchmarkArchitectureEmptyPatch(b *testing.B) {
	var assignments []qs.Assignment
	if architectureSparsePatch(assignments) != nil {
		b.Fatal("empty application patch constructed a query")
	}
	b.ReportAllocs()
	for b.Loop() {
		architectureStatement = architectureSparsePatch(assignments)
	}
}
